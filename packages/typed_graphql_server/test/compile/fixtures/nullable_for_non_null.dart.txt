import 'package:typed_graphql_server/typed_graphql_server.dart';

final class Pet {
  const Pet(this.name);
  final String? name;
}

void main() {
  final b = ExecutableSchemaBuilder<Object?>();
  b.object<Pet>(
    'Pet',
    fields: (t) => [t.field('name', b.string).pure((Pet p) => p.name)],
  );
}
