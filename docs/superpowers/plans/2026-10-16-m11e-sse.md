```dart
void main() {
  const parser = EventStreamParser(maxEventBytes: 1024);

    List<List<int>> chunks, [

      expect(pairs(await parse('event: next\ndata: 1\n\n')), [('next', '1')]);

    test('has the type message when the stream names none', () async {
      expect(pairs(await parse('data: 1\n\n')), [('message', '1')]);

      expect(pairs(await parse('data: a\ndata: b\ndata:\ndata: c\n\n')), [

      expect(pairs(await parse('data: a:b\n\n')), [('message', 'a:b')]);
```

```dart
import 'package:meta/meta.dart';

import 'event_stream_error.dart';
import 'server_sent_event.dart';

/// Splits decoded event-stream text into lines and lines into events,
/// counting the bytes it holds so a hostile stream cannot grow it (R0.3).
///
/// Only the lines that build the event (`data` and `event`) and the line
/// being read count; comments and ignored fields are dropped as soon as
/// their line ends, so a stream of them runs forever in constant memory.
@internal
final class EventAssembler({
  /// The most bytes one event, with the line being read, may hold.
  required final int maxEventBytes,
}) {
  /// Creates an assembler for one stream.
  this;

  static const int _lineFeed = 0x0A;
  static const int _carriageReturn = 0x0D;
  static const int _byteOrderMark = 0xFEFF;

  final StringBuffer _line = StringBuffer();
  final StringBuffer _data = StringBuffer();
  String? _type;
  bool _hasData = false;
  bool _isAtStart = true;
  bool _skipsLineFeed = false;
  int _lineBytes = 0;
  int _eventBytes = 0;

  /// Reads [text], the next piece of the stream, and returns the events it
  /// completes. Throws [EventStreamTooLarge] past the limit.
  List<ServerSentEvent> add(String text) {
    final events = <ServerSentEvent>[];
    for (final unit in text.codeUnits) {
      if (_isAtStart) {
        _isAtStart = false;
        if (unit == _byteOrderMark) continue;
      }
      if (_skipsLineFeed) {
        _skipsLineFeed = false;
        if (unit == _lineFeed) continue;
      }
      switch (unit) {
        case _carriageReturn:
          _skipsLineFeed = true;
          _endLine(events);
        case _lineFeed:
          _endLine(events);
        default:
          _append(unit);
      }
    }
    return events;
  }

  void _append(int unit) {
    _line.writeCharCode(unit);
    _lineBytes += _utf8Length(unit);
    if (_eventBytes + _lineBytes > maxEventBytes) {
      throw EventStreamTooLarge(maxEventBytes);
    }
  }

  /// The bytes a UTF-16 code unit takes in UTF-8; a surrogate pair is two
  /// units of two bytes, four in all.
  int _utf8Length(int unit) => switch (unit) {
    < 0x80 => 1,
    < 0x800 => 2,
    >= 0xD800 && <= 0xDFFF => 2,
    _ => 3,
  };

  void _endLine(List<ServerSentEvent> events) {
    final line = _line.toString();
    final bytes = _lineBytes;
    _line.clear();
    _lineBytes = 0;
    if (line.isEmpty) {
      _dispatch(events);
    } else if (!line.startsWith(':')) {
      _readField(line, bytes);
    }
  }

  void _readField(String line, int bytes) {
    final colon = line.indexOf(':');
    final field = colon < 0 ? line : line.substring(0, colon);
    var value = colon < 0 ? '' : line.substring(colon + 1);
    if (value.startsWith(' ')) {
      value = value.substring(1);
    }
    switch (field) {
      case 'data':
        if (_hasData) _data.write('\n');
        _data.write(value);
        _hasData = true;
      case 'event':
        _type = value;
      default:
        return;
    }
    _eventBytes += bytes + 1;
    if (_eventBytes > maxEventBytes) {
      throw EventStreamTooLarge(maxEventBytes);
    }
  }

  /// A blank line ends the event. It is dispatched when it has a type or a
  /// data line, which WHATWG does not do for an empty data buffer:
  /// graphql-sse's `complete` carries none.
  void _dispatch(List<ServerSentEvent> events) {
    final type = _type;
    if (type != null || _hasData) {
      events.add(
        ServerSentEvent(
          event: type == null || type.isEmpty ? 'message' : type,
          data: _data.toString(),
        ),
      );
    }
    _type = null;
    _data.clear();
    _hasData = false;
    _eventBytes = 0;
  }
}
```

```dart
import 'package:meta/meta.dart';

/// Why an event stream cannot be read further; the parser throws it for
/// hostile input and the SSE link turns it into a failure result.
@internal
sealed class EventStreamError implements Exception {
  /// Creates an error.
  const EventStreamError();
}

/// An event, or the line being read, is longer than the limit.
@internal
final class const EventStreamTooLarge(
  /// The limit in bytes.
  final int maxEventBytes,
) extends EventStreamError {
  /// Creates the error.
  this;

  @override
  String toString() => 'An event is longer than $maxEventBytes bytes.';
}

/// The stream's bytes are not UTF-8.
@internal
final class const EventStreamNotUtf8() extends EventStreamError {
  /// Creates the error.
  this;

  @override
  String toString() => 'The event stream is not UTF-8.';
}
```

```dart
import 'dart:async';
import 'dart:convert';

import 'package:meta/meta.dart';

import 'event_assembler.dart';
import 'event_stream_error.dart';
import 'server_sent_event.dart';

/// Reads an event stream from bytes as WHATWG HTML's "Interpreting an
/// event stream" describes, in bounded memory (spec section 9, R8.4).
///
/// The bytes are decoded as strict UTF-8 across chunk boundaries. The
/// stream of events fails with an [EventStreamError] for hostile input:
/// bytes that are not UTF-8 or an event longer than `maxEventBytes`; either
/// way the body is cancelled. An `Exception` of the body stream passes on
/// unchanged. An event the body ends in the middle of is dropped, as in a
/// browser.
@internal
final class const EventStreamParser({
  /// The most bytes one event, or the line being read, may hold.
  required final int maxEventBytes,
}) {
  /// Creates a parser.
  this;

  /// The events of [bytes], in order. The returned stream listens to
  /// [bytes] when it is listened to and cancels it when it is cancelled.
  Stream<ServerSentEvent> parse(Stream<List<int>> bytes) {
    final assembler = EventAssembler(maxEventBytes: maxEventBytes);
    final collector = _TextCollector();
    final decoder = utf8.decoder.startChunkedConversion(collector);
    return bytes.transform(
      StreamTransformer<List<int>, ServerSentEvent>.fromHandlers(
        handleData: (chunk, sink) {
          try {
            decoder.add(chunk);
          } on FormatException {
            throw const EventStreamNotUtf8();
          }
          assembler.add(collector.take()).forEach(sink.add);
        },
        handleDone: (sink) {
          try {
            decoder.close();
          } on FormatException {
            sink.addError(const EventStreamNotUtf8());
          }
          sink.close();
        },
      ),
    );
  }
}

/// The text the decoder has produced since the last [take].
final class _TextCollector implements Sink<String> {
  final StringBuffer _text = StringBuffer();

  @override
  void add(String data) => _text.write(data);

  @override
  void close() {}

  String take() {
    final text = _text.toString();
    _text.clear();
    return text;
  }
}
```

```dart
import 'package:meta/meta.dart';

/// One event of an event stream: its type and its data lines joined with
/// line feeds (WHATWG HTML, "Interpreting an event stream").
@internal
final class const ServerSentEvent({
  /// The `event` field, or `message` when the stream gave none.
  required final String event,

  /// The `data` lines, joined with `\n`; empty when there were none.
  required final String data,
}) {
  /// Creates an event.
```

````markdown
# Server-sent events

`SseLink` sends queries, mutations and subscriptions over server-sent
events, in graphql-sse's distinct connections mode: one HTTP request, one
event stream, one operation. It is the way to run subscriptions where a
WebSocket is not available or not wanted. Like `HttpLink`, it does not
depend on one HTTP library: it uses an `HttpTransport` (see the HTTP
guide), whose response body may arrive in pieces.

## SseLink

The program below answers a subscription from memory with an event stream
of two `next` events and a `complete`:

```dart

import 'package:typed_graphql_client/typed_graphql_client.dart';

final class TicksSubscription extends GraphQLSubscription<int> {
  const TicksSubscription();

  @override
  OperationDocument get document => const OperationDocument(
    name: 'Ticks',
    type: OperationType.subscription,
    text: 'subscription Ticks{ticks}',
    sha256: '14c970476cf383579b21ff6dbb8326a9f2e3ea340def21bcb5bf9cf9b1af1f38',
  );

  @override
  JsonObject get variablesJson => JsonObject({});

  @override
  DecodeResult<int> decodeData(JsonObject data) => switch (data['ticks']) {
    JsonNumber(:final asInt?) => DecodeSuccess(asInt),
    _ => const DecodeFailure('expected an integer', path: [PathKey('ticks')]),
  };
}

final class InMemoryServer implements HttpTransport {
  @override
  Future<TransportResponse> send(TransportRequest request) async {
    print('${request.method} ${request.uri} ${request.headers['accept']}');
    return TransportResponse(
      statusCode: 200,
      headers: {'content-type': 'text/event-stream'},
      body: Stream.fromIterable([
        utf8.encode('event: next\ndata: {"data":{"ticks":1}}\n\n'),
        utf8.encode('event: next\ndata: {"data":{"ticks":2}}\n\n'),
        utf8.encode('event: complete\ndata:\n\n'),
      ]),
    );
  }
}

    link: SseLink(
      uri: Uri.parse('https://example.com/graphql'),
      transport: InMemoryServer(),
    ),

    switch (response) {
      case GraphQLData(:final data):
        print('tick $data');
      case GraphQLErrors(:final errors):

## What the link sends

Every operation, a query or a mutation too, is a `POST` with the JSON body
`{"query": ..., "operationName": ..., "variables": ...}` and the headers
`accept: text/event-stream` and `content-type: application/json`. Headers
come from three places, each over the one before: the protocol's own, the
`headers` given to the link, and the `RequestHeaders` entries of the
request's context; names are compared in lower case.

A request with `UploadFiles` in its context is not sent: the link answers
`UnsupportedOperation` (`GQL468`). Send files over `HttpLink` with a
`SplitLink`.

Route subscriptions to `SseLink` and everything else to `HttpLink` with a
`SplitLink` (see the Links guide):

```text
SplitLink(
  test: (request) => request.type == OperationType.subscription,
  whenTrue: sseLink,
  whenFalse: httpLink,
)
```

## How it reads the stream

The body is parsed as an event stream (WHATWG HTML, "Interpreting an event
stream"), as UTF-8, across any split of the chunks. Each event is handled
as it arrives:

| Answer | Result |
|---|---|
| a `next` event | the GraphQL response in its data, with an `HttpResponseInfo` in the context |
| a `complete` event | the stream ends and the connection closes |
| an event of another type, or none | `EventStreamFailure` (`GQL473`), then the stream ends |
| the body ends before `complete` | `EventStreamFailure` (`GQL473`), then the stream ends |
| a `next` that is not a GraphQL response | `ResponseFormatFailure` (`GQL464`), then the stream ends |
| a 2xx event stream in a charset other than UTF-8 | `ResponseFormatFailure` (`GQL462`) |
| any other answer | read as `HttpLink` reads it |

A failure ends the stream and closes the connection. An answer that is
not an event stream, such as the `400` with a GraphQL response a server
sends for a bad request, is read by the rules of the HTTP guide: a GraphQL
response is a payload (so the client reports its errors), plain JSON with
a 2xx status is one result, and anything else is `GQL461` or `GQL462`. An
`Exception` of the transport or of the body stream is a `NetworkFailure`
(`GQL460`).

## Limits

An event, and the line being read, may hold at most `maxEventBytes` bytes
(16 MiB by default). Past it the link cancels the body and reports
`ResponseFormatFailure` (`GQL463`). Comments and the fields graphql-sse
does not use (`id`, `retry`) are dropped as soon as their line ends, so a
server that sends only keep-alive comments costs no memory. Bytes that are
not UTF-8 are `GQL464`, as is JSON nested deeper than `maxResponseDepth`
(512 by default).

## Cancelling

graphql-sse stops an operation by closing the connection. Cancelling the
result stream (or the call's `CancelToken`) completes the request's
`abortTrigger`, which the transport answers by closing the connection, and
cancels the body subscription. Pausing the stream pauses the body. The
link has no timeout, because a subscription lives as long as it is
listened to.

Single connection mode, where one stream carries every operation and
separate requests start them, is not supported.
````

```dart
/// ends the stream. A answer that is not an event stream is read as
/// [HttpLink] reads it, so a request the server refuses with a GraphQL
/// response is an errors response. An event is bounded by `maxEventBytes`.
/// Cancelling the result stream closes the connection, which is how the
/// protocol stops an operation. There is no timeout: a subscription lives
/// as long as the caller listens. Single connection mode is not supported.
/// A request with `UploadFiles` in its context is refused with
/// `UnsupportedOperation` (`GQL468`) without sending anything.
///
/// {@category sse}
final class SseLink extends Link {
  /// Creates a link that sends to [uri] through [transport], adding
  /// [headers] to every request; no event may pass [maxEventBytes], and the
  /// JSON of an event may nest [maxResponseDepth] levels.
  SseLink({
    required Uri uri,
    required HttpTransport transport,
    Map<String, String> headers = const {},
    int maxEventBytes = defaultMaxEventBytes,
    int maxResponseDepth = JsonReader.defaultMaxDepth,
  }) : _uri = uri,
       _encoder = HttpRequestEncoder(
         uri: uri,
         headers: headers,
         usesGetForQueries: false,
         random: Random(),
       ),
       _exchange = SseExchange(
         transport: transport,
         reader: ResponseReader(
           maxBytes: maxEventBytes,
           maxDepth: maxResponseDepth,
         ),
         parser: EventStreamParser(maxEventBytes: maxEventBytes),
       ) {
    RangeError.checkNotNegative(maxEventBytes, 'maxEventBytes');
    RangeError.checkValueInInterval(
      maxResponseDepth,
      1,
      JsonReader.maxSupportedDepth,
      'maxResponseDepth',
    );
  }

  /// The default of `maxEventBytes`: 16 MiB.
  static const int defaultMaxEventBytes = 16 * 1024 * 1024;

  static const Map<String, String> _protocolHeaders = {
    'accept': 'text/event-stream',
    'content-type': 'application/json',
  };

  final Uri _uri;
  final HttpRequestEncoder _encoder;
  final SseExchange _exchange;

  @override
  Stream<LinkResult> request(LinkRequest request, NextLink forward) {
    if (request.context.entry<UploadFiles>() != null) {
      return Stream.value(
        LinkFailure(
          UnsupportedOperation(
            .operationNotSupportedByLink,
            'SseLink cannot send files with ${request.document.name}; send '
            'it over HttpLink with a SplitLink.',
          ),
        ),
      );
    }
    return _exchange.run((abortTrigger) {
      final body = utf8.encode(jsonEncode(request.body.toJson().toJson()));
      return TransportRequest(
        method: 'POST',
        uri: _uri,
        headers: _encoder.headersFor(request, _protocolHeaders),
        body: Stream.value(body),
        contentLength: body.length,
        abortTrigger: abortTrigger,
      );
    });
  }
}
```

```dart
import 'dart:async';

import 'package:meta/meta.dart';

import '../client/client_failure.dart';
import '../http/http_transport.dart';
import '../http/media_type.dart';
import '../http/response_reader.dart';
import '../http/transport_request.dart';
import '../http/transport_response.dart';
import '../link/http_response_info.dart';
import '../link/link_context.dart';
import '../link/link_result.dart';
import 'event_stream_error.dart';
import 'event_stream_parser.dart';
import 'server_sent_event.dart';

/// One request of `SseLink`: sends it, then relays the answer to
/// the stream of [open] until the server completes, a failure comes or the
/// caller cancels. It ends the connection, through the abort trigger,
/// whenever it stops listening to the answer.
@internal
final class SseRun({
  /// Sends the request.
  required final HttpTransport transport,
  required final ResponseReader reader,
  required final EventStreamParser parser,
}) {
  /// Creates the run of one request.
  this;

  final StreamController<LinkResult> _controller =
      StreamController<LinkResult>();
  final Completer<void> _abort = Completer<void>();
  StreamSubscription<ServerSentEvent>? _events;
  bool _isFinished = false;

  /// The results of the request [encode] writes (given the request's abort
  /// trigger), sent when the stream is listened to. Call it once.
  Stream<LinkResult> open(
    TransportRequest Function(Future<void> abortTrigger) encode,
  ) {
    _controller
      ..onListen = (() => unawaited(_send(encode)))
      ..onPause = (() => _events?.pause())
      ..onResume = (() => _events?.resume())
      ..onCancel = _stop;
    return _controller.stream;
  }

  /// The listener cancelled: ends the connection and the body.
  Future<void> _stop() async {
    _isFinished = true;
    _release();
    await _events?.cancel();
  }

  Future<void> _send(
    TransportRequest Function(Future<void> abortTrigger) encode,
  ) async {
    final TransportResponse response;
    try {
      response = await transport.send(encode(_abort.future));
    } on Exception catch (error) {
      _finish(LinkFailure(NetworkFailure(error)));
      return;
    }
    if (_isFinished) return;
    final type = MediaType.parse(response.headers['content-type']);
    final isStream =
        response.statusCode >= 200 &&
        response.statusCode < 300 &&
        type.essence == MediaType.eventStream;
    if (!isStream) {
      await _readSingle(response);
    } else if (!type.isUtf8) {
      _finish(
        LinkFailure(
          ResponseFormatFailure(
            .unsupportedResponseType,
            'The response is in charset "${type.charset}"; only UTF-8 is '
            'read.',
          ),
        ),
      );
    } else {
      _listen(response);
    }
  }

  /// An answer that is not an event stream is one result, read as
  /// `HttpLink` reads it.
  Future<void> _readSingle(TransportResponse response) async {
    try {
      _finish(await reader.read(response));
    } on Exception catch (error) {
      _finish(LinkFailure(NetworkFailure(error)));
    }
  }

  void _listen(TransportResponse response) {
    final context = LinkContext([
      HttpResponseInfo(response.statusCode, headers: response.headers),
    ]);
    _events = parser
        .parse(response.body)
        .listen(
          (event) => _onEvent(event, context),
          onError: _onError,
          onDone: () => _finish(
            const LinkFailure(
              EventStreamFailure(
                'The event stream ended without a complete event.',
              ),
            ),
          ),
          cancelOnError: true,
        );
  }

  void _onEvent(ServerSentEvent event, LinkContext context) {
    switch (event.event) {
      case 'next':
        final result = reader.resultOf(event.data, context);
        if (result is LinkFailure) {
          _finish(result);
        } else {
          _controller.add(result);
        }
      case 'complete':
        _finish();
      case final other:
        _finish(
          LinkFailure(
            EventStreamFailure(
              'The event stream sent an event of type "$other", which '
              'graphql-sse does not define.',
            ),
          ),
        );
    }
  }

  void _onError(Object error, StackTrace stackTrace) {
    switch (error) {
      case EventStreamTooLarge():
        _finish(
          LinkFailure(
            ResponseFormatFailure(.responseTooLarge, error.toString()),
          ),
        );
      case EventStreamNotUtf8():
        _finish(
          LinkFailure(
            ResponseFormatFailure(.malformedResponse, error.toString()),
          ),
        );
      case Exception():
        _finish(LinkFailure(NetworkFailure(error)));
      default:
        if (!_isFinished) _controller.addError(error, stackTrace);
        _finish();
    }
  }

  /// Ends the run, after [last] if there is one; the first end wins.
  void _finish([LinkResult? last]) {
    if (_isFinished) return;
    _isFinished = true;
    if (last != null) _controller.add(last);
    unawaited(_events?.cancel());
    _release();
    unawaited(_controller.close());
  }

  /// Tells the transport the answer is no longer read; harmless once the
  /// body has ended.
  void _release() {
    if (!_abort.isCompleted) {
      _abort.complete();
    }
  }
}
```

## Review Focus

- The WHATWG rules: CR, LF and CRLF across chunk boundaries, the BOM only at the very start, comments, `data` joined with `\n`, dispatch only of an event with a type or data, `id` and `retry` ignored.
- Bounded memory: only `data` and `event` lines count toward `maxEventBytes`, per event; a line that never ends fails and cancels the body; endless comments run in constant space.
- Cancellation: cancelling the result aborts the transport request and cancels the body subscription even when the body is stalled (a transformer, not `async*`); pausing pauses the body.
- Every outcome is typed: unknown event or no `complete` `GQL473`, too large `GQL463`, not UTF-8 or not a response `GQL464`, another charset or media type `GQL462`, a non-2xx status `GQL461`, a body error `GQL460` after the events before it, files `GQL468`.
- Known quirk, accepted: events completed in the same chunk before an oversized line are dropped when the limit error ends the stream.
